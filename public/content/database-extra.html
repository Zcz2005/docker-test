sql/views/001_trace_views.sql
-- 完整溯源视图：聚合批次 + 溯源 + 质检
CREATE OR REPLACE VIEW v_batch_trace_full AS
SELECT
    b.batch_id,
    b.product_name,
    b.category,
    b.quantity,
    b.unit,
    b.status,
    o.name AS org_name,
    o.org_type,
    tr.record_id,
    tr.stage,
    tr.action,
    tr.data_hash,
    tr.operated_at,
    u.real_name AS operator_name,
    qr.result AS quality_result,
    qr.report_hash,
    bt.tx_id AS chain_tx_id,
    bt.status AS chain_status
FROM product_batches b
JOIN organizations o ON b.org_id = o.id
LEFT JOIN trace_records tr ON b.batch_id = tr.batch_id
LEFT JOIN users u ON tr.operator_id = u.id
LEFT JOIN quality_reports qr ON b.batch_id = qr.batch_id
LEFT JOIN blockchain_transactions bt ON b.chain_tx_id = bt.tx_id
ORDER BY b.batch_id, tr.operated_at;

-- 消费者公开查询视图（脱敏）
CREATE OR REPLACE VIEW v_public_trace AS
SELECT
    b.batch_id,
    b.product_name,
    b.status,
    tr.stage,
    tr.action,
    o.name AS org_name,
    tr.operated_at,
    tr.data_hash,
    qr.result AS quality_result
FROM product_batches b
JOIN trace_records tr ON b.batch_id = tr.batch_id
JOIN organizations o ON tr.org_id = o.id
LEFT JOIN quality_reports qr ON b.batch_id = qr.batch_id
WHERE b.status NOT IN ('RECALLED', 'DELETED');

-- 链上交易待确认监控视图
CREATE OR REPLACE VIEW v_pending_chain_tx AS
SELECT
    bt.tx_id,
    bt.chaincode,
    bt.function_name,
    bt.ref_table,
    bt.ref_id,
    bt.submitted_at,
    EXTRACT(EPOCH FROM (NOW() - bt.submitted_at)) AS pending_seconds
FROM blockchain_transactions bt
WHERE bt.status IN ('SUBMITTED', 'ENDORSED')
  AND bt.submitted_at < NOW() - INTERVAL '30 seconds';

触发器与自动审计

sql/triggers/001_audit_triggers.sql
-- 批次状态变更自动审计
CREATE OR REPLACE FUNCTION fn_audit_batch_status()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status IS DISTINCT FROM NEW.status THEN
        INSERT INTO audit_logs (action, resource_type, resource_id, request_data)
        VALUES (
            'BATCH_STATUS_CHANGE',
            'product_batches',
            NEW.batch_id,
            jsonb_build_object(
                'old_status', OLD.status,
                'new_status', NEW.status,
                'changed_at', NOW()
            )
        );
    END IF;
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_batch_status_audit
    BEFORE UPDATE ON product_batches
    FOR EACH ROW EXECUTE FUNCTION fn_audit_batch_status();

-- 链上交易确认后自动更新批次
CREATE OR REPLACE FUNCTION fn_sync_chain_status()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'COMMITTED' AND NEW.ref_table = 'product_batches' THEN
        UPDATE product_batches
        SET chain_status = 'CONFIRMED',
            chain_tx_id = NEW.tx_id,
            block_number = NEW.block_number
        WHERE batch_id = NEW.ref_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_chain_tx_sync
    AFTER UPDATE ON blockchain_transactions
    FOR EACH ROW EXECUTE FUNCTION fn_sync_chain_status();

sql/queries/business_queries.sql
-- ① 消费者溯源查询（与 API /public/trace 等价）
SELECT tr.stage, tr.action, o.name AS org, tr.operated_at, tr.data_hash
FROM trace_records tr
JOIN organizations o ON tr.org_id = o.id
WHERE tr.batch_id = 'GC-TOM-2026-SD-00421'
ORDER BY tr.operated_at ASC;

-- ② 召回影响范围分析（含子批次）
WITH RECURSIVE batch_tree AS (
    SELECT batch_id, parent_batch_id, quantity, status FROM product_batches
    WHERE batch_id = 'GC-TOM-2026-SD-00421'
    UNION ALL
    SELECT b.batch_id, b.parent_batch_id, b.quantity, b.status
    FROM product_batches b JOIN batch_tree bt ON b.parent_batch_id = bt.batch_id
)
SELECT * FROM batch_tree WHERE status = 'ON_SHELF';

-- ③ IoT 运输温度异常统计
SELECT
    batch_id,
    COUNT(*) FILTER (WHERE value > 8.0) AS alert_count,
    AVG(value) AS avg_temp,
    MIN(value) AS min_temp,
    MAX(value) AS max_temp
FROM iot_sensor_data
WHERE sensor_type = 'TEMPERATURE'
  AND batch_id = 'GC-TOM-2026-SD-00421'
  AND time BETWEEN '2026-06-04' AND '2026-06-05'
GROUP BY batch_id;

-- ④ 链上链下数据一致性校验
SELECT b.batch_id, b.chain_status,
       bt.status AS tx_status,
       CASE WHEN b.metadata_hash IS NOT NULL AND bt.status = 'COMMITTED'
            THEN 'CONSISTENT' ELSE 'CHECK_REQUIRED' END AS consistency
FROM product_batches b
LEFT JOIN blockchain_transactions bt ON b.chain_tx_id = bt.tx_id
WHERE b.chain_status = 'PENDING'
   OR (b.chain_status = 'CONFIRMED' AND bt.status != 'COMMITTED');

-- ⑤ 组织溯源操作排行（管理报表）
SELECT o.name, o.org_type,
       COUNT(tr.id) AS trace_count,
       COUNT(DISTINCT tr.batch_id) AS batch_count
FROM trace_records tr
JOIN organizations o ON tr.org_id = o.id
WHERE tr.operated_at >= NOW() - INTERVAL '30 days'
GROUP BY o.id ORDER BY trace_count DESC;
